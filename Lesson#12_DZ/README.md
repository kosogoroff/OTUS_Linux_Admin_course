## 1. Скрипт для запуска двух процессов с разными значениями nice

```
admin_insta11@mv334 ~]$ cat two_processes_9.sh
#!/usr/bin/env bash
set -u

VERBOSE=false    # подробный режим, статистика pidstat выводится дополнительно на экран
CYCLES=2000000   # количество выполняемых пустых циклов в нагрузочной функции
CPU=""           # равен номеру CPU, если ывполняется привязка обоих процессов к одному CPU
INTERVAL=1       # интервал сбора и вывода статистики pidstat
COUNT=50         # количество выводов статистики pidstat
NICE_HIGH=0      # значение nice первого процесса
NICE_LOW=0       # значение nice второго процесса

usage_help() {
  # Вывод подсказки по использованию скрипта
  cat >&2 << EOF1

Использование: $0 [опции] [<nice_high>] [<nice_low>]

     <nice_high> - значение nice (-20 - 19) для первого процесса (по умолчанию $NICE_HIGH)
     <nice_low> - значение nice (-20 - 19) для второго процесса (по умолчанию $NICE_LOW)

Внимание: для назначения отрицательных значений nice требуются права root - запуск скрипта через sudo

Опции: 
     -v|--verbose  - включение подробного вывода, 
     -c|--cycles <num> - количество циклов (по умолчанию $CYCLES), 
     -C|--cpu <num> - включает привязку обоих процессов к одному CPU с номером <num> (по умолчанию процессы не привязаны к CPU),
     -h|--help - вывод данной справки




Примеры использования скрипта:

$0               # без параметров выполняется с параметрами по умолчанию

$0 -10 10        # два нагрузочных процесса выполняются со значениями nice -10 и 10 

$0 -C 0          # два нагрузочных процесса выполняются с привязкой к одному CPU 0

$0 -C 0 -10 10   # два нагрузочных процесса выполняются с привязкой к одному CPU 0 и со значениями nice -10 и 10


EOF1

  exit 1
}

# разбор параметров
while [[ $# -gt 0 ]]; do
    case "$1" in
        -v|--verbose)
            VERBOSE=true
            shift
            ;;
        -c|--cycles)
            if [[ -z "$2" || "$2" =~ ^- ]]; then
                echo "Ошибка: после -c требуется число." >&2
                exit 1
            fi
            CYCLES="$2"
            shift 2
            ;;
        -C|--cpu)
            if [[ -z "$2" || "$2" =~ ^- ]]; then
                echo "Ошибка: после -C требуется номер CPU." >&2
                exit 1
            fi
            CPU="$2"
            shift 2
            ;;
        -h|--help)
            usage_help
            exit 0
            ;;
        *)
            break
            ;;
    esac
done

# последние два параметра - это nice процессов
if [[ $# -ge 1 ]]; then NICE_HIGH="$1"; fi
if [[ $# -ge 2 ]]; then NICE_LOW="$2"; fi

# проверяем значения nice
if ! [[ "$NICE_HIGH" =~  -20|^-?[1]{0,1}[0-9]$ ]] || ! [[ "$NICE_LOW" =~  -20|^-?[1]{0,1}[0-9]$ ]]; then
    echo "Ошибка: nice должны быть целыми числами." >&2
    exit 1
fi

echo "Запуск бенчмарка:"
echo "verbose = $VERBOSE"
echo "  nice_high = $NICE_HIGH"
echo "  nice_low  = $NICE_LOW"
echo "  циклы     = $CYCLES"
echo "  cpu        = ${CPU:-(без привязки)}"
echo

# --- НАГРУЗОЧНАЯ ФУНКЦИЯ С ЗАМЕРОМ ВРЕМЕНИ ---
run_burn() {
    local cycles=$1
    local nice_val=$2
    local cpu_val=$3

    # Замер старта в наносекундах
    local start_ns
    start_ns=$(date +%s%N)

    local i=0
    while (( i < cycles )); do
        ((i++))     #  нагрузочный цикл
        :
    done

    # Замер конца
    local end_ns
    end_ns=$(date +%s%N)

    # Вычисление длительности (scale=6 даёт 6 знаков после запятой)
    local duration_sec
    duration_sec=$(echo "scale=6; ($end_ns - $start_ns) / 1000000000" | bc -l)

    echo   
    echo "Выполнено $cycles циклов за ${duration_sec} сек (nice=$nice_val, cpu=${cpu_val:-any})" >&2
    echo
}

# объявляем в интерпретаторе функцию для её запуска через bash в качестве отдельного процесса
func_def=$(declare -f run_burn)

# запускаем первый процесс и получаем его PID
if [[ -n "$CPU" ]]; then
    # Если есть привязка процессов к определённому CPU, то запускаем через taskset с заданным nice
    taskset -c "$CPU" nice -n "$NICE_HIGH" bash -c "$func_def; run_burn $CYCLES $NICE_HIGH $CPU" &
else
    # Иначе запускаем без привязки к определённому CPU с заданным nice
    nice -n "$NICE_HIGH" bash -c "$func_def; run_burn $CYCLES $NICE_HIGH ''" &
fi
pid_high=$!

# запускаем второй процесс и получаем его PID
if [[ -n "$CPU" ]]; then
    # Если есть привязка процессов к определённому CPU, то запускаем через taskset с заданным nice
    taskset -c "$CPU" nice -n "$NICE_LOW" bash -c "$func_def; run_burn $CYCLES $NICE_LOW $CPU" &
else
    # Иначе запускаем без привязки к определённому CPU с заданным nice
    nice -n "$NICE_LOW" bash -c "$func_def; run_burn $CYCLES $NICE_LOW ''" &
fi
pid_low=$!

echo "PIDs: high=$pid_high, low=$pid_low"

# проверяем значения и psd/cpu и nice для двух запущенных процессов
ps -o pid,psr,ni,comm -p "$pid_high" "$pid_low"
echo

# Вывод статистики pidstat на экран только если нужен подробный режим
if [[ "$VERBOSE" == "true" ]]; then
     pidstat -u -p "$pid_high","$pid_low" "$INTERVAL" "$COUNT" | tee pidstat.log &
else
     # иначе статистика выводится только в файл pidstat.log
     pidstat -u -p "$pid_high","$pid_low" "$INTERVAL" "$COUNT" > pidstat.log &
fi
pidstat_pid=$!

wait "$pid_high" "$pid_low"
kill "$pidstat_pid" 2>/dev/null

echo "Статистика записана в файл pidstat.log"
echo "----------------------------------------"
```
